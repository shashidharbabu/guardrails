Commands to Run the Pipeline
# 1. Enter container and set up env
docker exec -it brave_diffie bash
cd /app/full_FinalMAD_with_judge
cp .env.example .env
# Edit .env: set VLLM_AGENTS_URL, VLLM_JUDGE_URL

# 2. Init DB and run debate + judge
python scripts/run_debate_from_inputs.py \
    --input examples/sample_input.json \
    --run-id demo-01 --db mad.db --run-judge

# 3. Inspect results
python scripts/inspect_db.py --db mad.db --run-id demo-01

# 4. Run judge separately (if debate ran without --run-judge)
python scripts/run_judge_on_db.py --run-id demo-01 --db mad.db

# 5. Export GRPO training data
python scripts/export_grpo_dataset.py --db mad.db --run-id demo-01 --out-dir grpo_export/
